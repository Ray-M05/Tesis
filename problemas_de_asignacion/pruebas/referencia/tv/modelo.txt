============================================================
MODELO DE OPTIMIZACION
============================================================

INDICES Y CONJUNTOS:
  B  =  conjunto de bloques      (|B| = 56)
  P  =  conjunto de programa disponibles   (|P| = 8)
  C  =  conjunto de canal   (|C| = 1)

VARIABLES DE DECISION:
  Para cada bloque t ∈ B se asigna:
    programa_t ∈ P   (programa)

FUNCION OBJETIVO:
  minimizar z(h) donde:
    z(h) = 1000 · Σ_i incumpl_i(h)   [duras]
         + 1 · Σ_j incumpl_j(h)   [blandas]

RESTRICCIONES DURAS  (nppq / tqpq):
  R1  [SIMPLE / DURA  (nppq) — No Puede Pasar Que]:
    adyacentes_mismo_tipo(h) = 0
    donde:
      adyacentes_mismo_tipo(h) = Σ_{c∈CANAL, e1∈EMISION, e2∈EMISION}  [((canal(programa(e1)) = c) ∧ (canal(programa(e2)) = c) ∧ (dia(bloque(e1)) = dia(bloque(e2))) ∧ (tipo(programa(e1)) = tipo(programa(e2))) ∧ ((orden(bloque(e1)) + 1) = orden(bloque(e2))))]


RESTRICCIONES BLANDAS  (tdpq / sbqp / minimizar):
  R2  [OPTIMIZACIÓN — Minimizar]:
    minimizar  max_{{x ∈ CANAL}}  consecutivos_por_canal(h, x)
    donde:
      consecutivos_por_canal(h, c) = Σ_{e1∈EMISION, e2∈EMISION}  [((canal(programa(e1)) = c) ∧ (canal(programa(e2)) = c) ∧ (dia(bloque(e1)) = dia(bloque(e2))) ∧ (tipo(programa(e1)) = tipo(programa(e2))) ∧ ((orden(bloque(e1)) + 1) = orden(bloque(e2))))]
