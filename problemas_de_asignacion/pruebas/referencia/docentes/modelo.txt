MODELO DE OPTIMIZACION
============================================================

INDICES Y CONJUNTOS:
  S  =  conjunto de sesions      (|S| = 70)
  A  =  conjunto de aula disponibles   (|A| = 6)
  D  =  conjunto de dia disponibles   (|D| = 5)
  T  =  conjunto de turno disponibles   (|T| = 6)
  P  =  conjunto de profesor   (|P| = 6)

VARIABLES DE DECISION:
  Para cada sesion t ∈ S se asigna:
    aula_t ∈ A   (aula)
    dia_t ∈ D   (dia)
    turno_t ∈ T   (turno)

FUNCION OBJETIVO:
  minimizar z(h) donde:
    z(h) = 1000 · Σ_i incumpl_i(h)   [duras]
         + 1 · Σ_j incumpl_j(h)   [blandas]

RESTRICCIONES DURAS  (nppq / tqpq):
  R1  [SIMPLE / DURA  (nppq) — No Puede Pasar Que]:
    colisiones_de_profesor(h) = 0
    donde:
      colisiones_de_profesor(h) = Σ_{p∈PROFESOR, c1∈CLASE, c2∈CLASE}  [((profesor(sesion(c1)) = p) ∧ (profesor(sesion(c2)) = p) ∧ (dia(c1) = dia(c2)) ∧ (turno(c1) = turno(c2)) ∧ (aula(c1) ≠ aula(c2)))]

  R2  [SIMPLE / DURA  (nppq) — No Puede Pasar Que]:
    colisiones_de_aula(h) = 0
    donde:
      colisiones_de_aula(h) = Σ_{c1∈CLASE, c2∈CLASE}  [((aula(c1) = aula(c2)) ∧ (dia(c1) = dia(c2)) ∧ (turno(c1) = turno(c2)) ∧ (grupo(sesion(c1)) ≠ grupo(sesion(c2))))]

  R3  [SIMPLE / DURA  (nppq) — No Puede Pasar Que]:
    colisiones_de_grupo(h) = 0
    donde:
      colisiones_de_grupo(h) = Σ_{c1∈CLASE, c2∈CLASE}  [((grupo(sesion(c1)) = grupo(sesion(c2))) ∧ (dia(c1) = dia(c2)) ∧ (turno(c1) = turno(c2)) ∧ (aula(c1) ≠ aula(c2)))]


RESTRICCIONES BLANDAS  (tdpq / sbqp / minimizar):
  R4  [OPTIMIZACIÓN — Minimizar]:
    minimizar  max_{{x ∈ PROFESOR}}  dias_de_profesor(h, x)
    donde:
      dias_de_profesor(h, prof) = |{ fecha(c) : c∈CLASE,  (profesor(sesion(c)) = prof) }|
